   259	/// Primary syscall entry point for routing `int 0x80` hardware traps.
   260	/// Interprets the `TrapFrame` register states and delegates to the appropriate
   261	/// ABI backend: native bootstrapping or the `PosixLcl` compatible wrapper.
   262	///
   263	/// This routing layer serves as the absolute lowest level in the HXNU architecture
   264	/// before hitting userspace. The Linux Compatibility Layer (LCL) daemon relies on
   265	/// the `PosixLcl` path to intercept `int 0x80` traps and securely translate them
   266	/// into bare-metal native HXNU/heterexec semantics, achieving zero-latency bridging.
   267	pub fn syscall_handler(trap_frame: &mut crate::arch::x86_64::SyscallRegisterFrame) -> SyscallOutcome {
   268	    let abi = match trap_frame.r12 {
   269	        2 => SyscallAbi::HxnuNativeBootstrap,
   270	        _ => SyscallAbi::PosixLcl,
   271	    };
   272	    match abi {
   273	        SyscallAbi::HxnuNativeBootstrap => dispatch_hxnu_bootstrap(
   274	            trap_frame.rax,
   275	            [
   276	                trap_frame.rdi,
   277	                trap_frame.rsi,
   278	                trap_frame.rdx,
   279	                trap_frame.r10,
   280	                trap_frame.r8,
   281	                trap_frame.r9,
   282	            ],
   283	        ),
   284	        SyscallAbi::PosixLcl => {
   285	            crate::tty::write_str("[LCL Handler] Intercepted int 0x80\n");
   286	            SyscallOutcome { value: 0, action: SyscallAction::YieldThread }
   287	        }
   288	        _ => SyscallOutcome::errno(ENOSYS),
   289	    }
   290	}
   291	
   292	pub fn dispatch(abi: SyscallAbi, number: u64, args: [u64; 6]) -> SyscallOutcome {
   293	    match abi {
   294	        SyscallAbi::LinuxBootstrap => posix_compat::dispatch_linux_bootstrap(number, args),
   295	        SyscallAbi::GhostBootstrap => posix_compat::dispatch_ghost_bootstrap(number, args),
   296	        SyscallAbi::HxnuNativeBootstrap => dispatch_hxnu_bootstrap(number, args),
   297	        SyscallAbi::PosixLcl => {
   298	            crate::tty::write_str("[LCL Handler] Intercepted int 0x80\n");
   299	            SyscallOutcome { value: 0, action: SyscallAction::YieldThread }
   300	        }
   301	        _ => SyscallOutcome::errno(ENOSYS),
   302	    }
   303	}

   348	    pub fn sys_stat(args: [u64; 6]) -> SyscallOutcome {
   349	        let ptr = args[0] as usize;
   350	        let statbuf = args[1] as usize;
   351	        let raw_path = match super::copyin_c_string(ptr, super::MAX_PATH_BYTES) {
   352	            Ok(p) => p,
   353	            Err(e) => return SyscallOutcome::errno(e),
   354	        };
   355	        let node = match crate::vfs::lookup(&raw_path) {
   356	            Some(n) => n,
   357	            None => return SyscallOutcome::errno(super::ENOENT),
   358	        };
   359	
   360	        let mut st_mode = 0o100644;
   361	        if node.kind == crate::vfs::VfsNodeKind::Directory {
   362	            st_mode = 0o040755;
   363	        }
   364	
   365	        let s = stat {
   366	            st_dev: 0, st_ino: 1, st_nlink: 1,
   367	            st_mode, st_uid: 0, st_gid: 0, __pad0: 0, st_rdev: 0,
   368	            st_size: node.size as i64, st_blksize: 4096, st_blocks: ((node.size + 4095) / 4096) as i64,
   369	            st_atime: 0, st_atime_nsec: 0, st_mtime: 0, st_mtime_nsec: 0, st_ctime: 0, st_ctime_nsec: 0,
   370	            __unused: [0; 3],
   371	        };
   372	
   373	        match super::copyout_struct(statbuf, &s) {
   374	            Ok(_) => SyscallOutcome::success(0),
   375	            Err(e) => SyscallOutcome::errno(e),
   376	        }
   377	    }

   379	    pub fn sys_lseek(args: [u64; 6]) -> SyscallOutcome {
   380	        let fd = args[0] as i32;
   381	        let offset = args[1] as i64;
   382	        let whence = args[2] as i32;
   383	
   384	        if fd < 0 || fd as usize >= crate::sched::MAX_PROCESS_FDS {
   385	            return SyscallOutcome::errno(super::EBADF);
   386	        }
   387	        let table = crate::sched::process_fd_table();
   388	        let global_idx = match table[fd as usize] {
   389	            Some(idx) => idx,
   390	            None => return SyscallOutcome::errno(super::EBADF),
   391	        };
   392	        let global_table = unsafe { &mut *super::FILE_TABLE.get() };
   393	        let open = match &mut global_table[global_idx] {
   394	            Some(f) => f,
   395	            None => return SyscallOutcome::errno(super::EBADF),
   396	        };
   397	
   398	        let new_offset = match whence {
   399	            0 => offset, // SEEK_SET
   400	            1 => open.offset as i64 + offset, // SEEK_CUR
   401	            2 => open.content.len() as i64 + offset, // SEEK_END
   402	            _ => return SyscallOutcome::errno(super::EINVAL),
   403	        };
   404	
   405	        if new_offset < 0 {
   406	            return SyscallOutcome::errno(super::EINVAL);
   407	        }
   408	
   409	        open.offset = new_offset as usize;
   410	        SyscallOutcome::success(new_offset)
   411	    }

   425	    pub fn sys_pipe(args: [u64; 6]) -> SyscallOutcome {
   426	        let ptr = args[0] as usize;
   427	        let global_table = unsafe { &mut *super::FILE_TABLE.get() };
   428	
   429	        let mut r_idx = -1;
   430	        let mut w_idx = -1;
   431	        
   432	        let mut r_fd = -1;
   433	        let mut w_fd = -1;
   434	
   435	        for i in 0..super::MAX_SYSTEM_FILES {
   436	            if global_table[i].is_none() {
   437	                if r_idx == -1 {
   438	                    r_idx = i as i32;
   439	                } else if w_idx == -1 {
   440	                    w_idx = i as i32;
   441	                    break;
   442	                }
   443	            }
   444	        }
   445	
   446	        if w_idx == -1 {
   447	            return SyscallOutcome::errno(super::EMFILE);
   448	        }
   449	
   450	        let mut table = crate::sched::process_fd_table();
   451	        for i in 3..crate::sched::MAX_PROCESS_FDS {
   452	            if table[i].is_none() {
   453	                if r_fd == -1 {
   454	                    r_fd = i as i32;
   455	                } else if w_fd == -1 {
   456	                    w_fd = i as i32;
   457	                    break;
   458	                }
   459	            }
   460	        }
   461	
   462	        if w_fd == -1 {
   463	            return SyscallOutcome::errno(super::EMFILE);
   464	        }
   465	
   466	        global_table[r_idx as usize] = Some(super::OpenFile {
   467	            ref_count: 1, path: alloc::string::String::from("pipe:read"), offset: 0, content: alloc::vec::Vec::new()
   468	        });
   469	        global_table[w_idx as usize] = Some(super::OpenFile {
   470	            ref_count: 1, path: alloc::string::String::from("pipe:write"), offset: 0, content: alloc::vec::Vec::new()
   471	        });
   472	
   473	        table[r_fd as usize] = Some(r_idx as usize);
   474	        table[w_fd as usize] = Some(w_idx as usize);
   475	        crate::sched::set_process_fd_table(table);
   476	
   477	        let fds = [r_fd, w_fd];
   478	        match super::copyout_struct(ptr, &fds) {
   479	            Ok(_) => SyscallOutcome::success(0),
   480	            Err(e) => SyscallOutcome::errno(e),
   481	        }
   482	    }

   484	    pub fn sys_dup(args: [u64; 6]) -> SyscallOutcome {
   485	        let oldfd = args[0] as i32;
   486	        if oldfd < 0 || oldfd as usize >= crate::sched::MAX_PROCESS_FDS {
   487	            return SyscallOutcome::errno(super::EBADF);
   488	        }
   489	        let mut table = crate::sched::process_fd_table();
   490	        let global_idx = match table[oldfd as usize] {
   491	            Some(idx) => idx,
   492	            None => return SyscallOutcome::errno(super::EBADF),
   493	        };
   494	
   495	        let mut newfd = -1;
   496	        for i in 3..crate::sched::MAX_PROCESS_FDS {
   497	            if table[i].is_none() {
   498	                newfd = i as i32;
   499	                break;
   500	            }
   501	        }
   502	        if newfd == -1 {
   503	            return SyscallOutcome::errno(super::EMFILE);
   504	        }
   505	
   506	        table[newfd as usize] = Some(global_idx);
   507	        let global_table = unsafe { &mut *super::FILE_TABLE.get() };
   508	        if let Some(open) = &mut global_table[global_idx] {
   509	            open.ref_count += 1;
   510	        }
   511	        crate::sched::set_process_fd_table(table);
   512	
   513	        SyscallOutcome::success(newfd as i64)
   514	    }

   516	    pub fn sys_dup2(args: [u64; 6]) -> SyscallOutcome {
   517	        let oldfd = args[0] as i32;
   518	        let newfd = args[1] as i32;
   519	        if oldfd < 0 || oldfd as usize >= crate::sched::MAX_PROCESS_FDS || newfd < 0 || newfd as usize >= crate::sched::MAX_PROCESS_FDS {
   520	            return SyscallOutcome::errno(super::EBADF);
   521	        }
   522	        if oldfd == newfd {
   523	            return SyscallOutcome::success(newfd as i64);
   524	        }
   525	
   526	        let mut table = crate::sched::process_fd_table();
   527	        let global_idx = match table[oldfd as usize] {
   528	            Some(idx) => idx,
   529	            None => return SyscallOutcome::errno(super::EBADF),
   530	        };
   531	
   532	        if let Some(old_g_idx) = table[newfd as usize] {
   533	            let global_table = unsafe { &mut *super::FILE_TABLE.get() };
   534	            if let Some(open) = &mut global_table[old_g_idx] {
   535	                if open.ref_count > 1 {
   536	                    open.ref_count -= 1;
   537	                } else {
   538	                    global_table[old_g_idx] = None;
   539	                }
   540	            }
   541	        }
   542	
   543	        table[newfd as usize] = Some(global_idx);
   544	        let global_table = unsafe { &mut *super::FILE_TABLE.get() };
   545	        if let Some(open) = &mut global_table[global_idx] {
   546	            open.ref_count += 1;
   547	        }
   548	        crate::sched::set_process_fd_table(table);
   549	
   550	        SyscallOutcome::success(newfd as i64)
   551	    }

   553	    pub fn sys_fork(_args: [u64; 6]) -> SyscallOutcome {
   554	        match crate::sched::sys_fork_current_thread() {

   560	    pub fn sys_wait4(args: [u64; 6]) -> SyscallOutcome {
   561	        let pid = args[0] as i64;
   562	        let status_ptr = args[1] as usize;
   563	        let options = args[2] as u32;
   564	
   565	        let parent_pid = crate::sched::current_process_id();
   566	        if pid == -1 {
   567	            loop {
   568	                match crate::sched::sys_reap_child(parent_pid) {
   569	                    Ok((reaped_pid, status)) => {
   570	                        if status_ptr != 0 {
   571	                            let raw_status = status << 8;
   572	                            let _ = super::copyout_struct(status_ptr, &raw_status);
   573	                        }
   574	                        return SyscallOutcome::success(reaped_pid as i64);
   575	                    }
   576	                    Err(has_children) => {
   577	                        if !has_children {
   578	                            return SyscallOutcome::errno(super::ENOENT);
   579	                        }
   580	                        if (options & 1) != 0 {
   581	                            return SyscallOutcome::success(0);
   582	                        }
   583	                        return super::SyscallOutcome { value: 0, action: super::SyscallAction::YieldThread };
   584	                    }
   585	                }
   586	            }
   587	        } else {
   588	            return SyscallOutcome::errno(super::ENOSYS);
   589	        }
   590	    }

   892	fn open_path_at(dirfd: i64, path_ptr: usize, flags: u64) -> SyscallOutcome {
   893	    if !is_read_only_open(flags) {
   894	        return SyscallOutcome::errno(EINVAL);
   895	    }
   896	
   897	    let raw_path = match copyin_c_string(path_ptr, MAX_PATH_BYTES) {
   898	        Ok(path) => path,
   899	        Err(error) => return SyscallOutcome::errno(error),
   900	    };
   901	    if raw_path.is_empty() {
   902	        return SyscallOutcome::errno(EINVAL);
   903	    }
   904	
   905	    let resolved_path = if raw_path.starts_with('/') {
   906	        raw_path
   907	    } else if dirfd == AT_FDCWD {
   908	        let mut absolute = String::from("/");
   909	        absolute.push_str(&raw_path);
   910	        absolute
   911	    } else {
   912	        return SyscallOutcome::errno(ENOSYS);
   913	    };
   914	
   915	    let node = match vfs::lookup(&resolved_path) {
   916	        Some(node) => node,
   917	        None => return SyscallOutcome::errno(ENOENT),
   918	    };
   919	    if node.kind == VfsNodeKind::Directory {
   920	        return SyscallOutcome::errno(EISDIR);
   921	    }
   922	
   923	    let content = match vfs::read(&node.path) {
   924	        Some(content) => content.into_bytes(),
   925	        None => return SyscallOutcome::errno(EIO),
   926	    };
   927	    match alloc_open_file(node.path, content) {
   928	        Ok(fd) => SyscallOutcome::success(fd),
   929	        Err(error) => SyscallOutcome::errno(error),
   930	    }
   931	}

   933	fn read_from_fd(args: [u64; 6]) -> SyscallOutcome {
   934	    let fd = match parse_fd(args[0]) {
   935	        Ok(fd) => fd,
   936	        Err(error) => return SyscallOutcome::errno(error),
   937	    };
   938	
   939	    let count = match usize::try_from(args[2]) {
   940	        Ok(count) => count,
   941	        Err(_) => return SyscallOutcome::errno(ERANGE),
   942	    };
   943	    if count > MAX_READ_BYTES {
   944	        return SyscallOutcome::errno(ERANGE);
   945	    }
   946	
   947	    match read_open_file(fd, args[1] as usize, count) {
   948	        Ok(read) => SyscallOutcome::success(read),
   949	        Err(error) => SyscallOutcome::errno(error),
   950	    }
   951	}

   977	fn write_text(ptr: usize, len: u64) -> SyscallOutcome {
   978	    let count = match usize::try_from(len) {
   979	        Ok(count) => count,
   980	        Err(_) => return SyscallOutcome::errno(ERANGE),
   981	    };
   982	    if count > MAX_WRITE_BYTES {
   983	        return SyscallOutcome::errno(ERANGE);
   984	    }
   985	    if count == 0 {
   986	        return SyscallOutcome::success(0);
   987	    }
   988	
   989	    let bytes = match copyin_bytes(ptr, count) {
   990	        Ok(bytes) => bytes,
   991	        Err(error) => return SyscallOutcome::errno(error),
   992	    };
   993	
   994	    let text = sanitize_for_console(&bytes);
   995	    tty::write_str(&text);
   996	
   997	    match i64::try_from(count) {
   998	        Ok(written) => SyscallOutcome::success(written),
   999	        Err(_) => SyscallOutcome::errno(ERANGE),
  1000	    }
  1001	}

  1078	fn sanitize_for_console(bytes: &[u8]) -> String {
  1079	    let mut text = String::with_capacity(bytes.len());
  1080	    for &byte in bytes {
  1081	        match byte {
  1082	            b'\n' | b'\r' | b'\t' | 0x20..=0x7e => text.push(byte as char),
  1083	            _ => text.push('?'),
  1084	        }
  1085	    }
  1086	    text
  1087	}

  1102	    let mut table = sched::process_fd_table();
  1103	    let mut fd = -1;
  1104	    for i in 3..sched::MAX_PROCESS_FDS {
  1105	        if table[i].is_none() {
  1106	            fd = i as i32;
  1107	            break;
  1108	        }
  1109	    }
  1110	    if fd == -1 {
  1111	        return Err(EMFILE);
  1112	    }
  1113	    
  1114	    let global_table = unsafe { &mut *FILE_TABLE.get() };
  1115	    let mut global_idx = -1;

  1117	        if global_table[i].is_none() {
  1118	            global_table[i] = Some(OpenFile {
  1119	                ref_count: 1,
  1120	                path: path.clone(),
  1121	                offset: 0,

  1123	            });
  1124	            global_idx = i as i32;

  1126	        }
  1127	    }
  1128	    if global_idx == -1 {

  1130	    }

  1137	fn read_open_file(fd: i32, destination_ptr: usize, count: usize) -> Result<i64, i64> {
  1138	    if fd < 0 || fd as usize >= sched::MAX_PROCESS_FDS {
  1139	        return Err(EBADF);
  1140	    }
  1141	    let table = sched::process_fd_table();
  1142	    let global_idx = table[fd as usize].ok_or(EBADF)?;
  1143	    
  1144	    let global_table = unsafe { &mut *FILE_TABLE.get() };
  1145	    let open = global_table[global_idx].as_mut().ok_or(EBADF)?;
  1146	    
  1147	    if count == 0 {
  1148	        return Ok(0);
  1149	    }
  1150	
  1151	    let available = open.content.len().saturating_sub(open.offset);
  1152	    let read_len = min(count, available);
  1153	    if read_len > 0 {
  1154	        let bytes = &open.content[open.offset..open.offset + read_len];
  1155	        uaccess::copyout(bytes, destination_ptr).map_err(map_uaccess_error)?;
  1156	        open.offset = open.offset.saturating_add(read_len);
  1157	    }
  1158	    i64::try_from(read_len).map_err(|_| ERANGE)
  1159	}

  1200	fn copyin_c_string(ptr: usize, max_len: usize) -> Result<String, i64> {
  1201	    let mut bytes = Vec::new();

  1203	        let address = ptr.checked_add(index).ok_or(ERANGE)?;
  1204	        let mut byte = [0u8; 1];

  1208	            return Ok(String::from(text));

  1210	        bytes.push(byte[0]);
  1211	    }

  1213	    Err(ERANGE)
